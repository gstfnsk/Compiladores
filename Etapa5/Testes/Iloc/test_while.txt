// Objetivo: Testar a geração de código para a estrutura de repetição 'enquanto'.
//
// Comportamento esperado:
// 1. 'i' começa em 0, 'sum' começa em 0.
// 2. O loop 'enquanto' executa enquanto 'i < 5'.
// 3. A cada iteração, 'sum' é incrementado pelo valor de 'i', e 'i' é incrementado em 1.
//    - i=0, sum=0
//    - i=1, sum=1
//    - i=2, sum=3
//    - i=3, sum=6
//    - i=4, sum=10
// 4. Quando 'i' se torna 5, a condição 'i < 5' é falsa e o loop termina.
//
// Valores finais esperados na memória: i=5, sum=10.

main -> inteiro := [
  var i := inteiro com 0
  var sum := inteiro com 0
  enquanto (i < 5) [
    sum := sum + i
    i := i + 1
  ]
];
