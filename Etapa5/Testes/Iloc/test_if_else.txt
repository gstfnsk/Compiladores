// Objetivo: Testar a geração de código para estruturas de controle de fluxo 'se' e 'se-senao'.
// Também testa operadores relacionais (>, <, ==).
//
// Comportamento esperado:
// 1. 'a' = 5, 'b' = 10, 'c' = 0.
// 2. Primeira condição 'se (a > b)' é falsa. O bloco não executa.
// 3. Segunda condição 'se (a < b)' é verdadeira.
//    - 'c' se torna 1.
//    - O 'senao' é ignorado.
// 4. Terceira condição 'se (c == 1)' é verdadeira.
//    - 'b' se torna 20.
//    - O 'senao' é ignorado.
//
// Valores finais esperados na memória: a=5, b=20, c=1.

main -> inteiro := [
  var a := inteiro com 5
  var b := inteiro com 10
  var c := inteiro com 0

  se (a > b)
  [ 
    c := 100
  ]

  se (a < b)
  [
    c := 1
  ]
  senao
  [
    c := 2
  ]

  se (c == 1)
  [
    b := 20
  ]
  senao
  [
    b := 30
  ]
];
